```java id="x54mra"
import javax.swing.JOptionPane;

public class Apresentacao {

    public static void main(String[] args) {

        String nome = "Octavio";
        String profissional = "Estudante de ADS";
        String sistemaOperacional = "NixOS";

        String linguagens = "MySQL, C, C++, Python";
        String[] idiomas = {"pt_BR"};

        StringBuilder message = new StringBuilder();

        message.append("Olá! Eu sou ").append(nome);
        message.append(", um ").append(profissional);
        message.append(".\n\n");

        message.append("💻 Sistema Operacional: ")
               .append(sistemaOperacional);

        message.append("\n👨‍💻 Tecnologias: ")
               .append(linguagens);

        message.append("\n🌎 Idiomas:");

        for (String idioma : idiomas) {
            message.append("\n- ").append(idioma);
        }

        message.append(
            "\n\nObrigado por visitar meu perfil!\n" +
            "Espero que goste do meu trabalho 🚀"
        );

        JOptionPane.showMessageDialog(
            null,
            message.toString(),
            "Apresentação",
            JOptionPane.INFORMATION_MESSAGE
        );
    }
}
```
